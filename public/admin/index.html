TypeScript SDK Quickstart (OmniSocketClient)

import { OmniSocketClient } from './sdk/omni-client';

const client = new OmniSocketClient({
  url: 'http://localhost:3000',
  applicationId: 'YOUR_APP_SLUG',
  apiKey: 'YOUR_TENANT_API_KEY',
  userId: 'user_alice'
});

await client.connect();

// Join room & broadcast signal
await client.joinRoom('conference-room-101');
await client.broadcastToRoom('conference-room-101', 'signal:webrtc_offer', { sdp: 'v=0...' });
